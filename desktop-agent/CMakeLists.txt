cmake_minimum_required(VERSION 3.20)
project(daylight_agent VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The companion requires Windows APIs.")
endif()
add_executable(daylight-agent src/main.cpp src/activity.cpp src/http.cpp)
target_include_directories(daylight-agent PRIVATE include)
target_compile_features(daylight-agent PRIVATE cxx_std_17)
target_compile_definitions(daylight-agent PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0a00)
target_link_libraries(daylight-agent PRIVATE user32 winhttp rpcrt4)
if(MSVC)
  target_compile_options(daylight-agent PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(daylight-agent PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME agent-self-test COMMAND daylight-agent --self-test)
add_test(NAME opt-in-required COMMAND daylight-agent --once)
set_tests_properties(opt-in-required PROPERTIES WILL_FAIL TRUE)
